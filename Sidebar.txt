        document.querySelectorAll('.tab-btn').forEach(function(b) { b.classList.remove('active'); });
        document.querySelectorAll('.tab-content').forEach(function(t) { t.classList.remove('active'); });
        btn.classList.add('active');
        document.getElementById('tab-' + btn.dataset.tab).classList.add('active');
      });
    });

    function showStatus(msg, isError) {
      var el = document.getElementById('status');
      el.textContent = msg;
      el.className = 'status visible' + (isError ? ' error' : '');
      setTimeout(function() { el.className = 'status'; }, 3000);
    }

    // TEMPLATE INSERTION

    function runTemplate(funcName) {
      showStatus('Building template...');
      google.script.run
        .withSuccessHandler(function(result) {
          showStatus(result.message || 'Template created.');
        })
        .withFailureHandler(function(err) {
          showStatus('Error: ' + err.message, true);
        })
        [funcName]();
    }

    function runFunction(funcName) {
      google.script.run
        .withSuccessHandler(function() { showStatus('Done.'); })
        .withFailureHandler(function(err) { showStatus('Error: ' + err.message, true); })
        [funcName]();
    }

    // ============================================
    // BRANDING
    // ============================================
    function getBrandFromForm() {
      return {
        name: document.getElementById('brandName').value.trim(),
        orgName: document.getElementById('orgName').value.trim(),
        primaryColor: document.getElementById('primaryColor').value,
        accentColor: document.getElementById('accentColor').value,
        headingFont: document.getElementById('headingFont').value,
        bodyFont: document.getElementById('bodyFont').value
      };
    }

    function populateForm(profile) {
      document.getElementById('brandName').value = profile.name || '';
      document.getElementById('orgName').value = profile.orgName || '';
      document.getElementById('primaryColor').value = profile.primaryColor || '#1B3A5C';
      document.getElementById('accentColor').value = profile.accentColor || '#C0392B';
      document.getElementById('headingFont').value = profile.headingFont || 'Georgia';
      document.getElementById('bodyFont').value = profile.bodyFont || 'Arial';
    }

    function saveBrand() {
      var data = getBrandFromForm();
      if (!data.name) { showStatus('Enter a profile name.', true); return; }
      google.script.run
        .withSuccessHandler(function() {
          showStatus('Profile "' + data.name + '" saved.');
          loadProfiles();
        })
        .withFailureHandler(function(err) { showStatus('Error: ' + err.message, true); })
        .saveBrandProfile(data);
    }

    function setActive() {
      var name = document.getElementById('brandName').value.trim();
      if (!name) { showStatus('Enter a profile name.', true); return; }
      // Save first, then set active
      var data = getBrandFromForm();
      google.script.run
        .withSuccessHandler(function() {
          google.script.run
            .withSuccessHandler(function() {
              showStatus('"' + name + '" is now the active brand.');
              loadProfiles();
            })
            .setActiveBrand(name);
        })
        .saveBrandProfile(data);
    }

    function loadProfiles() {
      google.script.run
        .withSuccessHandler(function(profiles) {
          var container = document.getElementById('profilesList');
          var keys = Object.keys(profiles);
          if (keys.length === 0) {
            container.innerHTML = '<p style="color: var(--text-muted); font-size: 11px;">No saved profiles yet.</p>';
            return;
          }
          var html = '';
          keys.forEach(function(key) {
            var p = profiles[key];
            html += '<div class="profile-item">' +
              '<div><div class="name">' + escapeHtml(p.name) + '</div>' +
              '<div class="colors">' +
              '<span class="swatch" style="background:' + p.primaryColor + '"></span>' +
              '<span class="swatch" style="background:' + p.accentColor + '"></span>' +
              '<span style="font-size:10px;color:var(--text-muted);">' + p.headingFont + ' / ' + p.bodyFont + '</span>' +
              '</div></div>' +
              '<div class="profile-actions">' +
              '<button onclick="loadProfile(\'' + escapeHtml(p.name) + '\')">Load</button>' +
              '<button class="delete" onclick="deleteProfile(\'' + escapeHtml(p.name) + '\')">Delete</button>' +
              '</div></div>';
          });
          container.innerHTML = html;
        })
        .loadBrandProfiles();
    }

    function loadProfile(name) {
      google.script.run
        .withSuccessHandler(function(profiles) {
          if (profiles[name]) {
            populateForm(profiles[name]);
            showStatus('Loaded "' + name + '".');
          }
        })
        .loadBrandProfiles();
    }

    function deleteProfile(name) {
      if (!confirm('Delete profile "' + name + '"?')) return;
      google.script.run
        .withSuccessHandler(function() {
          showStatus('Profile deleted.');
          loadProfiles();
        })
        .deleteBrandProfile(name);
    }

    function escapeHtml(str) {
      var div = document.createElement('div');
      div.textContent = str;
      return div.innerHTML;
    }

    // ============================================
    // INIT
    // ============================================
    loadProfiles();
